async function getCurrentProfile() { const { data: { user }, error: userError } = await supabaseClient.auth.getUser(); if (userError || !user) { console.error( "User error:", userError ); return null; } const { data: profile, error: profileError } = await supabaseClient .from("profiles") .select(` id, business_id, full_name, role, active `) .eq("id", user.id) .single(); if (profileError) { console.error( "Profile error:", profileError ); return null; } if (profile.active === false) { alert( "This POS user is disabled." ); return null; } return profile; }
